## apps/api/src/services/email.ts

```typescript
import sgMail from '@sendgrid/mail';
import { env } from '../config/env';

if (env.SENDGRID_API_KEY) {
  sgMail.setApiKey(env.SENDGRID_API_KEY);
}

export async function sendBreachMonitorEmail(
  to: string,
  breachSummary: { name: string; date: string; dataClasses: string[] }[]
): Promise<void> {
  if (!env.SENDGRID_API_KEY) {
    console.warn('SENDGRID_API_KEY not set — skipping breach monitor email');
    return;
  }

  const breachList = breachSummary.length > 0
    ? breachSummary.map(b =>
        `<li><strong>${b.name}</strong> (${b.date ?? 'unknown date'})<br>
         Exposed data: ${b.dataClasses.join(', ')}</li>`
      ).join('')
    : '<li>No new breaches detected this week. ✅</li>';

  await sgMail.send({
    to,
    from: { email: env.SENDGRID_FROM_EMAIL, name: 'DefendDaily SentryLife' },
    subject: `Your Weekly Security Report — ${new Date().toLocaleDateString()}`,
    html: `
      <div style="font-family: sans-serif; max-width: 600px; margin: 0 auto;">
        <h1 style="color: #7C3AED;">🛡️ SentryLife Weekly Breach Report</h1>
        <p>Here's your weekly breach monitoring summary for linked email addresses:</p>
        <ul>${breachList}</ul>
        <p style="color: #6B7280; font-size: 12px;">
          This report is generated by DefendDaily SentryLife. 
          Data sourced from HaveIBeenPwned.
        </p>
      </div>
    `,
  });
}
```

## apps/api/src/jobs/familyBreachMonitor.ts (Step 5.11 — integrates with HIBP job)

```typescript
import { Worker, Job } from 'bullmq';
import { Queue } from 'bullmq';
import { connection } from './queue';
import { db } from '../db/client';
import { checkEmailBreaches } from '../services/hibp';
import { sendBreachMonitorEmail } from '../services/email';

export const familyBreachQueue = new Queue('family-breach-monitor', { connection });

export async function scheduleFamilyBreachJob(): Promise<void> {
  const existing = await familyBreachQueue.getRepeatableJobs();
  for (const job of existing) await familyBreachQueue.removeRepeatableByKey(job.key);

  await familyBreachQueue.add('family-breach-scan', {}, {
    repeat: { cron: '0 7 * * 1', tz: 'UTC' }, // Monday 7 AM
    removeOnComplete: 5,
  });
}

export const familyBreachWorker = new Worker(
  'family-breach-monitor',
  async (_job: Job) => {
    // Get all family group primary users (employees who invited family)
    const primaries = await db.query<{
      id: string; email: string; family_group_id: string;
    }>(`
      SELECT DISTINCT u.id, u.email, u.family_group_id
      FROM users u
      WHERE u.family_group_id IS NOT NULL AND u.role = 'employee'
    `);

    for (const primary of primaries.rows) {
      try {
        // Get all family member emails in this group
        const familyResult = await db.query<{ email: string }>(
          `SELECT email FROM users WHERE family_group_id = $1 AND email NOT LIKE '%@slack.local'`,
          [primary.family_group_id]
        );

        const breachSummary: { name: string; date: string; dataClasses: string[] }[] = [];

        for (const member of familyResult.rows) {
          const breaches = await checkEmailBreaches(member.email);
          for (const breach of breaches) {
            breachSummary.push({
              name: breach.Name,
              date: breach.BreachDate,
              dataClasses: breach.DataClasses,
            });
          }
        }

        await sendBreachMonitorEmail(primary.email, breachSummary);
      } catch (err: unknown) {
        const message = err instanceof Error ? err.message : String(err);
        await db.query(
          `INSERT INTO audit_log (user_id, action, metadata) VALUES ($1, 'job_failure', $2)`,
          [primary.id, JSON.stringify({ job: 'family-breach-monitor', error: message })]
        );
      }
    }
  },
  { connection }
);
```
